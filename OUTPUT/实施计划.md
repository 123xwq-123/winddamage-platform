    return _page(_login_page_html("账号或密码错误" if error else ""))


@router.post("/login")
async def login_submit(request: Request):
    form = await request.form()
    status, payload = auth.authenticate(str(form.get("username") or ""),
                                        str(form.get("password") or ""))
    if status == "ok":
        resp = RedirectResponse("/screen", status_code=302)
        resp.set_cookie(auth.SESSION_COOKIE, auth.create_session(payload["id"]),
                        httponly=True, samesite="lax",
                        max_age=auth.SESSION_TTL_HOURS * 3600, path="/")
        return resp
    return _page(_login_page_html("账号或密码错误" if status != "locked"
                                 else f"已锁定，请 {payload} 分钟后重试"))
```

@router.get("/api/overview")
async def api_overview(request: Request):
    user = _current_user(request)
    scope = None if user and user.get("role") == "admin" else (user["id"] if user else 0)
    week = _count_week(scope)      # query_records(date_from=周一) 取 total
    recheck = history.get_stats(scope)["high_risk"]  # 待复检口径=高风险台账，MVP 同值展示
    return JSONResponse(_overview_payload(scope, week, recheck))


@router.post("/api/analyze")
async def api_analyze(request: Request):
    user = _current_user(request)
    if not user:
        return JSONResponse({"detail": "not authenticated"}, status_code=401)
    form = await request.form()
    up = form.get("file")
    if up is None or not getattr(up, "filename", ""):
        return JSONResponse({"detail": "no file"}, status_code=400)
    tmpd = tempfile.mkdtemp(prefix="sc_")
    path = os.path.join(tmpd, os.path.basename(up.filename))
    with open(path, "wb") as f:
        f.write(await up.read())
    try:
        from app import _do_inference, get_image_info
        parsed, raw, infer_time = _do_inference(path)
        if parsed is None:
            return JSONResponse({"detail": "inference failed", "raw": raw[:500]}, status_code=400)
        rid = history.save_record(os.path.basename(path), parsed, infer_time, path,
                                  source="screen", persist_image=True,
                                  owner_user_id=user["id"], created_by=user["username"])
        payload = _report_payload(parsed, rid, infer_time, get_image_info(path))
        payload["image_info"]["filename"] = os.path.basename(up.filename)
        return JSONResponse(payload)
    finally:
        shutil.rmtree(tmpd, ignore_errors=True)
```
